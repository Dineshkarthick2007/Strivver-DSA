import java.util.*;
public class Main
{
    public static int[] reverseArray(int[] arr,int start,int end)
    {
        if(start>end) return arr;
        int temp=arr[start];
        arr[start]=arr[end];
        arr[end]=temp;
        
        return reverseArray(arr,start+1,end-1);
        
    }
    public static void main(String[] args)
    {
        Scanner s = new Scanner(System.in);
        int n=s.nextInt();
        int[] arr = new int[n];
        for(int i=0;i<n;i++)
        {
            arr[i]=s.nextInt();
        }
        reverseArray(arr,0,arr.length-1);

        for(int i=0;i<arr.length;i++)
        {
            System.out.print(arr[i]+" ");
        }
    }
}

class Solution {
    public boolean PalindromeHelper(char[] str,int start,int end)
    {
        if(start>end) return true;
        if(str[start]!=str[end]) return false;

        return PalindromeHelper(str,start+1,end-1);
    }
    public boolean isPalindrome(String s) {
        String temp = s.replaceAll("[^a-zA-Z0-9]","").toLowerCase();
        char str[] = temp.toCharArray();
        

        if(PalindromeHelper(str,0,str.length-1)) return true;
        return false;
    }
}
