import java.util.*;
public class Main
{
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int n=s.nextInt();
		// Inverted centered pyramid 
		for(int i=n;i>=1;i--)
		{
		    for(int k=1;k<=n-i;k++)
		    {
		        System.out.print(" ");
		    }
		    for(int j=1;j<=2*i-1;j++)
		    {
		        System.out.print("*");
		    }
		    System.out.println();
		}
		
		// Centered pyramid
		for(int i=1;i<=n;i++)
		{
		    for(int k=1;k<=n-i;k++)
		    {
		        System.out.print(" ");
		    }
		    for(int j=1;j<=2*i-1;j++)
		    {
		        System.out.print("*");
		    }
		    System.out.println();
		}		
		
	}
}
