import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);

        int n;
        n = s.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = s.nextInt();
        }

        int[] hash = new int[100];
        for(int i=0;i<n;i++)
        {
            hash[arr[i]]+=1;
        }

        int q=s.nextInt();
        while(q--!=0)
        {
            int num=s.nextInt();
            System.out.print(hash[num]+" ");
        }
    }
}
