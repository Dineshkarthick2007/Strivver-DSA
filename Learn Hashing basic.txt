//Counting frequencies of each element using brute force approach

import java.util.*;
public class Main{
    public static void main(String[] args)
    {
    Scanner s = new Scanner(System.in);
    int n=s.nextInt();
    int[] arr = new int[n];

    for(int i=0;i<n;i++)
    {
        arr[i]=s.nextInt();
    }
    
    for(int i=0;i<n;i++)
    {
        int count=0;
        int temp=arr[i];
        for(int j=0;j<n;j++)
        {
            if(arr[j]==temp)
            {
                count++;
                arr[j]=-1;
            }
            
        }
        if(temp!=-1)
        {
            System.out.println(temp+"-->"+count);
        }
            
    }
    }
}


class Solution {

    public int maxFrequency(int[] nums, int k) {

        Arrays.sort(nums);

        int answer = 1;

        for (int x = nums.length - 1; x >= 0; x--) {

            int max = nums[x];
            int tempK = k;
            int count = 1;

            for (int i = x - 1; i >= 0; i--) {

                int diff = max - nums[i];

                if (tempK >= diff) {
                    tempK = tempK - diff;
                    count++;
                }
                else {
                    break;
                }
            }

            answer = Math.max(answer, count);
        }

        return answer;
    }
}
