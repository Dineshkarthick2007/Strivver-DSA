import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);

        int n;
        n = s.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = s.nextInt();
        }

        int[] hash = new int[100];
        for(int i=0;i<n;i++)
        {
            hash[arr[i]]+=1;
        }

        int q=s.nextInt();
        while(q--!=0)
        {
            int num=s.nextInt();
            System.out.print(hash[num]+" ");
        }
    }
}

import java.util.*;
public class Main{
    public static void main(String[] args)
    {
        Scanner s = new Scanner(System.in);
        String str=s.next();
        
        int[] hash = new int[256];

        for(int i=0;i<str.length();i++)
        {
            hash[str.charAt(i)]++;
        }

        int q=s.nextInt();
        while(q-- !=0)
        {
            char ch = s.next().charAt(0);
            System.out.println(hash[ch]);
        }
    }
